Reading & Writing Files

File I/O Operations

# WRITING to a file
file = open("output.txt", "w")  # w = write mode
file.write("Hello, World!")
file.write("\nSecond line")
file.close()

# BETTER: using "with" (auto-closes the file)
with open("output.txt", "w") as f:
    f.write("Line 1\n")
    f.write("Line 2\n")

# READING from a file
with open("output.txt", "r") as f:  # r = read mode
    content = f.read()      # Read entire file
    print(content)

# Read line by line
with open("output.txt", "r") as f:
    for line in f:
        print(line.strip())

# Read all lines into a list
with open("output.txt", "r") as f:
    lines = f.readlines()  # ["Line 1\n", "Line 2\n"]

# Append mode ("a")
with open("output.txt", "a") as f:
    f.write("New line added\n")
CSV Handling

Working with CSV Data

import csv

# READING a CSV file
with open("students.csv", "r") as f:
    reader = csv.reader(f)
    for row in reader:
        print(row)  # ["Alice", "95", "A"]

# WRITING a CSV file
data = [
    ["Name", "Score", "Grade"],
    ["Alice", 95, "A"],
    ["Bob", 82, "B"],
    ["Charlie", 78, "C"]
]

with open("output.csv", "w", newline="") as f:
    writer = csv.writer(f)
    writer.writerows(data)

# Using DictReader for named columns
with open("students.csv") as f:
    reader = csv.DictReader(f)
    for row in reader:
        print(row["Name"], row["Score"])
Exception Handling

Writing Robust Programs

# BASIC try/except
try:
    num = int(input("Enter a number: "))
    result = 100 / num
    print("Result:", result)
except ValueError:
    print("That's not a valid number!")
except ZeroDivisionError:
    print("Cannot divide by zero!")
except Exception as e:
    print("Something went wrong:", e)

# try/except/else/finally
try:
    f = open("data.txt")
    data = f.read()
except FileNotFoundError:
    print("File not found!")
else:
    print("File loaded successfully")
finally:
    print("This always runs")

# Practical: safe file reader
def safe_read(filename):
    try:
        with open(filename) as f:
            return f.read()
    except FileNotFoundError:
        print(filename + " not found")
        return None
