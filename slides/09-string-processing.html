String Methods

Transforming Text

# Case conversion
text = "Hello, World!"
print(text.upper())       # HELLO, WORLD!
print(text.lower())       # hello, world!
print(text.title())       # Hello, World!

# Whitespace handling
data = "  Hello, World!  "
print(data.strip())       # "Hello, World!"
print(data.lstrip())      # "Hello, World!  "

# Search and replace
sentence = "I love Python. Python is great!"
print(sentence.count("Python"))  # 2
print(sentence.find("Python"))   # 7

new = sentence.replace("Python", "Java")
print(new)  # "I love Java. Java is great!"

# Split and join
words = sentence.split()
print(words)  # ["I", "love", "Python.", "Python", "is", "great!"]

back = " - ".join(words)
print(back)
f-strings & Formatting

Beautiful Output

# Using .format() method
name = "Alice"
age = 20
gpa = 3.85

msg = "Name: {}, Age: {}, GPA: {}".format(name, age, gpa)
print(msg)  # Name: Alice, Age: 20, GPA: 3.85

# Named placeholders
print("{n} is {a} years old".format(n=name, a=age))

# Format specifiers
pi = 3.14159265
print("{:.2f}".format(pi))   # 3.14
print("{:.4f}".format(pi))   # 3.1416

# Padding and alignment
print("{:>10}".format("right"))   #     right
print("{:<10}".format("left"))    # left
print("{:^10}".format("center"))  #   center
print("{:*^10}".format("hi"))    # ****hi****

# String slicing
s = "Python Programming"
print(s[0:6])     # "Python"
print(s[7:])      # "Programming"
print(s[::-1])    # reversed
String Libraries & Validation

Built-in String Tools

import string
print(string.ascii_lowercase)  # abcdefghijklmnopqrstuvwxyz
print(string.digits)           # 0123456789

# Common validation patterns
phone = "1234567890"
print(phone.isdigit())    # True

name = "Alice"
print(name.isalpha())     # True

username = "alice2024"
print(username.isalnum()) # True

# String repetition
line = "=" * 40
print(line)  # ========================================

# Practical: format a student report card
students = [("Alice", 95), ("Bob", 82), ("Charlie", 78)]
header = "{:<15} {:>5}".format("Name", "Score")
print(header)
print("-" * 20)
for name, score in students:
    print("{:<15} {:>5}".format(name, score))
