for Loop & range()

Iterating with Precision

# Basic for loop -- iterate over a list
fruits = ["apple", "banana", "cherry"]
for fruit in fruits:
    print(fruit)
# Output: apple, banana, cherry

# range() generates sequences of numbers
for i in range(5):        # 0, 1, 2, 3, 4
    print(i)

for i in range(1, 6):     # 1, 2, 3, 4, 5
    print(i)

for i in range(0, 20, 3): # 0, 3, 6, 9, 12, 15, 18
    print(i)

# Practical: sum of numbers 1 to 100
total = 0
for i in range(1, 101):
    total += i
print("Sum = " + str(total))  # Sum = 5050

# Iterate over a string
for letter in "Python":
    print(letter, end=" ")
# P y t h o n
while Loop

Repeating Until Done

# while loop -- keeps going while condition is True
count = 5
while count > 0:
    print("Countdown:", count)
    count -= 1
print("Liftoff!")

# Input validation with while
while True:
    age = input("Enter your age: ")
    if age.isdigit():
        age = int(age)
        if 0 < age < 150:
            break  # Exit the loop
    print("Invalid! Try again.")
print("Your age is:", age)

# break -- exit the loop immediately
for i in range(1, 100):
    if i == 5:
        break      # Stop at 5
    print(i)       # 1, 2, 3, 4

# continue -- skip to next iteration
for i in range(1, 11):
    if i % 2 == 0:
        continue   # Skip even numbers
    print(i)       # 1, 3, 5, 7, 9
Nested Loops & Patterns

Loops Inside Loops

# Multiplication table (1-5)
for i in range(1, 6):
    for j in range(1, 6):
        print(i * j, end="\t")
    print()  # New line after each row

# Star pattern
for i in range(1, 6):
    print("*" * i)
# *
# **
# ***
# ****
# *****

# Find primes from 1 to 20
for num in range(2, 21):
    is_prime = True
    for i in range(2, num):
        if num % i == 0:
            is_prime = False
            break
    if is_prime:
        print(num, end=" ")
# 2 3 5 7 11 13 17 19
Warning: Nested loops multiply complexity. A loop with 1000 iterations inside another 1000-iteration loop runs 1,000,000 times. Keep inner loops small when possible.
